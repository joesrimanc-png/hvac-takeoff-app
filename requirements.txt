from __future__ import annotations

import os
import sqlite3
import tkinter as tk
from pathlib import Path
from tkinter import filedialog, messagebox, ttk

from openpyxl import Workbook
from reportlab.lib import colors
from reportlab.lib.pagesizes import letter
from reportlab.lib.styles import getSampleStyleSheet
from reportlab.platypus import Paragraph, SimpleDocTemplate, Spacer, Table, TableStyle

from hvac_takeoff.calculations import CATEGORY_ORDER, calculate_line_total, currency, summarize_items, unit_label
from hvac_takeoff.database import add_item, create_project, delete_item, get_project, initialize_db, list_items, list_projects, touch_project


def project_dir() -> Path:
    return Path.cwd()


class HVACTakeoffApp(tk.Tk):
    def __init__(self) -> None:
        super().__init__()
        initialize_db()
        self.title("HVAC Takeoff Pro")
        self.geometry("1400x900")
        self.minsize(1220, 760)
        self.current_project_id = None
        self.current_project_name = tk.StringVar(value="Main HVAC Estimate")
        self.unit_var = tk.StringVar(value="Imperial")
        self.category_var = tk.StringVar(value="Duct")
        self.project_list = []

        self.configure(bg="#f4f6fb")
        self._build_ui()
        self.load_projects()

    def _build_ui(self) -> None:
        top = tk.Frame(self, bg="#f4f6fb", padx=18, pady=14)
        top.pack(fill="x")

        tk.Label(top, text="Project Name", font=("Segoe UI", 11, "bold"), bg="#f4f6fb").grid(row=0, column=0, sticky="w")
        self.project_name_entry = tk.Entry(top, width=28, font=("Segoe UI", 11))
        self.project_name_entry.insert(0, "Main HVAC Estimate")
        self.project_name_entry.grid(row=0, column=1, padx=(8, 16), pady=6)

        tk.Label(top, text="Unit System", font=("Segoe UI", 11, "bold"), bg="#f4f6fb").grid(row=0, column=2, sticky="w")
        self.unit_combo = ttk.Combobox(top, values=["Imperial", "Metric"], width=12, state="readonly")
        self.unit_combo.set("Imperial")
        self.unit_combo.grid(row=0, column=3, padx=(8, 16), pady=6)

        self.create_project_button = tk.Button(top, text="Create Project", command=self.create_project, bg="#1a73e8", fg="white", font=("Segoe UI", 10, "bold"), padx=14, pady=8)
        self.create_project_button.grid(row=0, column=4, padx=(0, 12))

        self.export_pdf_button = tk.Button(top, text="Export PDF", command=self.export_pdf, bg="#00796b", fg="white", font=("Segoe UI", 10, "bold"), padx=12, pady=8)
        self.export_pdf_button.grid(row=0, column=5, padx=(0, 12))

        self.export_excel_button = tk.Button(top, text="Export Excel", command=self.export_excel, bg="#5b2c83", fg="white", font=("Segoe UI", 10, "bold"), padx=12, pady=8)
        self.export_excel_button.grid(row=0, column=6)

        content = tk.Frame(self, bg="#f4f6fb")
        content.pack(fill="both", expand=True, padx=18, pady=(0, 18))

        left = tk.Frame(content, bg="#ffffff", bd=1, relief="solid", padx=16, pady=14)
        left.pack(side="left", fill="y", padx=(0, 18))
        left.config(width=300)

        tk.Label(left, text="Project Dashboard", font=("Segoe UI", 14, "bold"), bg="#ffffff").pack(anchor="w")
        tk.Label(left, text="Select project", font=("Segoe UI", 10, "bold"), bg="#ffffff").pack(anchor="w", pady=(12, 8))

        self.project_listbox = tk.Listbox(left, height=18, width=28, font=("Segoe UI", 10), exportselection=False)
        self.project_listbox.pack(fill="both", expand=True)
        self.project_listbox.bind("<<ListboxSelect>>", self.on_project_selected)

        right = tk.Frame(content, bg="#f4f6fb")
        right.pack(side="right", fill="both", expand=True)

        summary_frame = tk.Frame(right, bg="#f4f6fb")
        summary_frame.pack(fill="x", pady=(0, 12))

        self.cards = {}
        for i, label in enumerate(["Material", "Labor", "Total", "Items"]):
            card = tk.Frame(summary_frame, bg="#ffffff", bd=1, relief="solid", padx=16, pady=12)
            card.grid(row=0, column=i, sticky="nsew", padx=(0 if i == 0 else 12, 0), ipadx=18)
            summary_frame.grid_columnconfigure(i, weight=1)
            tk.Label(card, text=label, font=("Segoe UI", 10, "bold"), bg="#ffffff").pack(anchor="w")
            value = tk.Label(card, text="$0.00", font=("Segoe UI", 18, "bold"), bg="#ffffff", fg="#1a73e8")
            value.pack(anchor="w", pady=(8, 0))
            self.cards[label] = value

        form = tk.Frame(right, bg="#ffffff", bd=1, relief="solid", padx=16, pady=16)
        form.pack(fill="x", pady=(0, 12))

        fields = [
            ("Category", "category"),
            ("Description", "description"),
            ("Qty", "quantity"),
            ("Unit", "unit"),
            ("Unit Cost", "unit_cost"),
            ("Labor Rate", "labor_rate"),
            ("Adjustment Factor", "adjustment_factor"),
            ("Notes", "notes"),
        ]
        self.item_fields = {}
        col = 0
        row = 0
        for label_text, key in fields:
            tk.Label(form, text=label_text, font=("Segoe UI", 10, "bold"), bg="#ffffff").grid(row=row, column=col, sticky="w", padx=(0, 10), pady=6)
            if key == "category":
                field = ttk.Combobox(form, values=CATEGORY_ORDER, width=18, state="readonly")
                field.set("Duct")
            elif key == "quantity":
                field = tk.Entry(form, width=18)
            elif key == "unit":
                field = tk.Entry(form, width=18)
                field.insert(0, "LF")
            else:
                field = tk.Entry(form, width=18)
            field.grid(row=row, column=col + 1, padx=(0, 16), pady=6)
            self.item_fields[key] = field
            col += 2
            if col >= 6:
                col = 0
                row += 1

        row += 1
        add_button = tk.Button(form, text="Add Estimate Item", command=self.add_item, bg="#228B22", fg="white", font=("Segoe UI", 10, "bold"), padx=16, pady=8)
        add_button.grid(row=row, column=0, columnspan=2, sticky="w", pady=(12, 6))

        table_frame = tk.Frame(right, bg="#ffffff", bd=1, relief="solid")
        table_frame.pack(fill="both", expand=True)

        columns = ("id", "category", "description", "qty", "unit", "unit_cost", "labor_rate", "adjust_factor", "material", "labor", "total", "notes")
        self.tree = ttk.Treeview(table_frame, columns=columns, show="headings")
        self.tree.heading("id", text="ID")
        self.tree.heading("category", text="Category")
        self.tree.heading("description", text="Description")
        self.tree.heading("qty", text="Qty")
        self.tree.heading("unit", text="Unit")
        self.tree.heading("unit_cost", text="Unit Cost")
        self.tree.heading("labor_rate", text="Labor Rate")
        self.tree.heading("adjust_factor", text="Factor")
        self.tree.heading("material", text="Material")
        self.tree.heading("labor", text="Labor")
        self.tree.heading("total", text="Total")
        self.tree.heading("notes", text="Notes")

        for col in columns:
            self.tree.column(col, width=90, anchor="center")
            if col in {"description", "notes"}:
                self.tree.column(col, width=150, anchor="w")

        self.tree.pack(fill="both", expand=True, padx=12, pady=12)

        actions = tk.Frame(table_frame, bg="#ffffff")
        actions.pack(fill="x", padx=12, pady=(0, 12))
        tk.Button(actions, text="Delete Selected", command=self.delete_item, bg="#d93025", fg="white", font=("Segoe UI", 10, "bold"), padx=12, pady=8).pack(side="left")
        tk.Button(actions, text="Refresh", command=self.refresh_project_view, bg="#f1b500", fg="black", font=("Segoe UI", 10, "bold"), padx=12, pady=8).pack(side="left", padx=(12, 0))

    def create_project(self) -> None:
        name = self.project_name_entry.get().strip() or "Main HVAC Estimate"
        unit_system = self.unit_combo.get() or "Imperial"
        project_id = create_project(name, unit_system)
        self.current_project_id = project_id
        self.current_project_name.set(name)
        self.project_name_entry.delete(0, tk.END)
        self.project_name_entry.insert(0, name)
        self.unit_combo.set(unit_system)
        self.load_projects()
        self.refresh_project_view()

    def load_projects(self) -> None:
        self.project_list = list_projects()
        self.project_listbox.delete(0, tk.END)
        for project in self.project_list:
            label = f"{project['name']} ({project['unit_system']})"
            self.project_listbox.insert(tk.END, label)
        if self.project_list:
            if self.current_project_id is None:
                self.current_project_id = self.project_list[0]["id"]
            self.project_listbox.selection_clear(0, tk.END)
            for index, project in enumerate(self.project_list):
                if project["id"] == self.current_project_id:
                    self.project_listbox.select_set(index)
                    break

    def on_project_selected(self, _event) -> None:
        selection = self.project_listbox.curselection()
        if not selection:
            return
        selected = self.project_list[selection[0]]
        self.current_project_id = selected["id"]
        self.project_name_entry.delete(0, tk.END)
        self.project_name_entry.insert(0, selected["name"])
        self.unit_combo.set(selected["unit_system"])
        self.refresh_project_view()

    def add_item(self) -> None:
        if not self.current_project_id:
            messagebox.showwarning("No Project", "Create or select a project first.")
            return

        try:
            qty = float(self.item_fields["quantity"].get() or 0)
            unit_cost = float(self.item_fields["unit_cost"].get() or 0)
            labor_rate = float(self.item_fields["labor_rate"].get() or 0)
            factor = float(self.item_fields["adjustment_factor"].get() or 1.0)
        except ValueError:
            messagebox.showerror("Invalid Input", "Quantity, unit cost, labor rate, and adjustment factor must be numeric.")
            return

        add_item(
            project_id=self.current_project_id,
            category=self.item_fields["category"].get(),
            description=self.item_fields["description"].get(),
            quantity=qty,
            unit=self.item_fields["unit"].get() or "EA",
            unit_cost=unit_cost,
            labor_rate=labor_rate,
            adjustment_factor=factor,
            notes=self.item_fields["notes"].get(),
        )
        touch_project(self.current_project_id)
        self.refresh_project_view()
        self.clear_item_form()

    def clear_item_form(self) -> None:
        self.item_fields["description"].delete(0, tk.END)
        self.item_fields["quantity"].delete(0, tk.END)
        self.item_fields["unit"].delete(0, tk.END)
        self.item_fields["unit"].insert(0, "LF")
        self.item_fields["unit_cost"].delete(0, tk.END)
        self.item_fields["labor_rate"].delete(0, tk.END)
        self.item_fields["adjustment_factor"].delete(0, tk.END)
        self.item_fields["adjustment_factor"].insert(0, "1")
        self.item_fields["notes"].delete(0, tk.END)
        self.item_fields["category"].set("Duct")

    def delete_item(self) -> None:
        selected = self.tree.selection()
        if not selected:
            messagebox.showwarning("No selection", "Select an item to remove.")
            return
        item_id = int(self.tree.item(selected[0], "values")[0])
        delete_item(item_id)
        touch_project(self.current_project_id)
        self.refresh_project_view()

    def refresh_project_view(self) -> None:
        if self.current_project_id is None:
            return

        project = get_project(self.current_project_id)
        if not project:
            return

        items = list_items(self.current_project_id)
        summary = summarize_items(items)

        self.cards["Material"].config(text=currency(summary["material"]))
        self.cards["Labor"].config(text=currency(summary["labor"]))
        self.cards["Total"].config(text=currency(summary["total"]))
        self.cards["Items"].config(text=str(summary["items"]))

        for row in self.tree.get_children():
            self.tree.delete(row)

        for item in items:
            line_totals = calculate_line_total(
                item["quantity"],
                item["unit_cost"],
                item["labor_rate"],
                item["adjustment_factor"],
            )
            self.tree.insert(
                "",
                tk.END,
                values=(
                    item["id"],
                    item["category"],
                    item["description"],
                    item["quantity"],
                    item["unit"],
                    currency(item["unit_cost"]),
                    currency(item["labor_rate"]),
                    item["adjustment_factor"],
                    currency(line_totals["material_total"]),
                    currency(line_totals["labor_total"]),
                    currency(line_totals["total"]),
                    item["notes"],
                ),
            )

    def export_excel(self) -> None:
        if self.current_project_id is None:
            messagebox.showwarning("No Project", "Select a project to export.")
            return

        items = list_items(self.current_project_id)
        project = get_project(self.current_project_id)
        path = filedialog.asksaveasfilename(
            initialfile=f"{project['name']}.xlsx",
            defaultextension=".xlsx",
            filetypes=[("Excel Workbook", "*.xlsx")],
        )
        if not path:
            return

        workbook = Workbook()
        summary = workbook.active
        summary.title = "Summary"
        summary.append(["Project", project["name"]])
        summary.append(["Unit System", project["unit_system"]])
        summary.append([])
        summary.append(["Item Count", len(items)])
        data = summarize_items(items)
        summary.append(["Material Total", currency(data["material"])])
        summary.append(["Labor Total", currency(data["labor"])])
        summary.append(["Overall Total", currency(data["total"])])

        lines = workbook.create_sheet("Line Items")
        lines.append(["ID", "Category", "Description", "Qty", "Unit", "Unit Cost", "Labor Rate", "Factor", "Material", "Labor", "Total", "Notes"])
        for item in items:
            totals = calculate_line_total(item["quantity"], item["unit_cost"], item["labor_rate"], item["adjustment_factor"])
            lines.append([
                item["id"],
                item["category"],
                item["description"],
                item["quantity"],
                item["unit"],
                item["unit_cost"],
                item["labor_rate"],
                item["adjustment_factor"],
                totals["material_total"],
                totals["labor_total"],
                totals["total"],
                item["notes"],
            ])

        workbook.save(path)
        messagebox.showinfo("Export Complete", f"Export created at:\n{path}")

    def export_pdf(self) -> None:
        if self.current_project_id is None:
            messagebox.showwarning("No Project", "Select a project to export.")
            return

        project = get_project(self.current_project_id)
        items = list_items(self.current_project_id)
        summary = summarize_items(items)

        path = filedialog.asksaveasfilename(
            initialfile=f"{project['name']}.pdf",
            defaultextension=".pdf",
            filetypes=[("PDF Document", "*.pdf")],
        )
        if not path:
            return

        story = []
        styles = getSampleStyleSheet()
        title = Paragraph(f"HVAC Takeoff Report - {project['name']}", styles["Title"])
        story.append(title)
        story.append(Spacer(1, 18))
        story.append(Paragraph(f"Unit System: {project['unit_system']} | Default Unit: {unit_label(project['unit_system'])}", styles["BodyText"]))
        story.append(Paragraph(f"Material Total: {currency(summary['material'])}", styles["BodyText"]))
        story.append(Paragraph(f"Labor Total: {currency(summary['labor'])}", styles["BodyText"]))
        story.append(Paragraph(f"Overall Total: {currency(summary['total'])}", styles["BodyText"]))
        story.append(Spacer(1, 14))

        table_data = [["Category", "Description", "Qty", "Unit", "Material", "Labor", "Total"]]
        for item in items:
            totals = calculate_line_total(item["quantity"], item["unit_cost"], item["labor_rate"], item["adjustment_factor"])
            table_data.append([
                item["category"],
                item["description"],
                str(item["quantity"]),
                item["unit"],
                currency(totals["material_total"]),
                currency(totals["labor_total"]),
                currency(totals["total"]),
            ])

        table = Table(table_data, colWidths=[70, 180, 50, 50, 90, 90, 90])
        table.setStyle(
            TableStyle([
                ("BACKGROUND", (0, 0), (-1, 0), colors.HexColor("#1a73e8")),
                ("TEXTCOLOR", (0, 0), (-1, 0), colors.whitesmoke),
                ("GRID", (0, 0), (-1, -1), 0.8, colors.grey),
                ("VALIGN", (0, 0), (-1, -1), "MIDDLE"),
                ("ROWBACKGROUNDS", (0, 1), (-1, -1), [colors.whitesmoke, colors.white]),
            ])
        )
        story.append(table)

        doc = SimpleDocTemplate(path, pagesize=letter)
        doc.build(story)
        messagebox.showinfo("Export Complete", f"PDF created at:\n{path}")


def main() -> None:
    app = HVACTakeoffApp()
    app.mainloop()
